Total Sales =
SUM(FactInternetSales[SalesAmount])
    + SUM(FactResellerSales[SalesAmount])


Age Group = 
VAR Age =
    DATEDIFF(
        DimCustomer[BirthDate],
        DATE(2014, 1, 28),
        YEAR
    )
RETURN
    SWITCH(
        TRUE(),
        Age >= 20 && Age < 25, "20-24",
        Age >= 25 && Age < 30, "25-29",
        Age >= 30 && Age < 35, "30-34",
        Age >= 35 && Age < 40, "35-39",
        Age >= 40 && Age < 45, "40-44",
        Age >= 45 && Age < 50, "45-49",
        "50+"
    )

AOV = 
DIVIDE(
    [Total Sales],
    DISTINCTCOUNT(FactInternetSales[SalesOrderNumber])
        + DISTINCTCOUNT(FactResellerSales[SalesOrderNumber])
)

Average Sales per Customer = 
DIVIDE(
    SUM(FactInternetSales[SalesAmount]),
    DISTINCTCOUNT(FactInternetSales[CustomerKey])
)


Average Customer Spending = 
DIVIDE(
    SUM(FactInternetSales[SalesAmount]),
    DISTINCTCOUNT(FactInternetSales[CustomerKey])
)
